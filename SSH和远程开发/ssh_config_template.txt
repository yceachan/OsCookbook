# SSH Client Configuration Template

# Host-specific settings
# Replace 'your_alias' with a short, memorable name for your connection
Host your_alias
    # Replace 'your_server_ip_or_hostname' with the actual IP address or hostname of the remote server
    Hostname your_server_ip_or_hostname
    
    # Replace 'your_username' with your username on the remote server
    User your_username
    
    # Optional: Specify the path to your private key file.
    # Replace '~/.ssh/id_rsa' with the actual path to your private key.
    # If you are using an SSH agent, this might not be necessary.
    # IdentityFile ~/.ssh/id_rsa
    
    # Optional: Specify the port number if it's not the default (22)
    # Port 22
    
    # Optional: Enable SSH agent forwarding. Useful if you need to access other servers
    # from the remote server using your local SSH keys.
    # ForwardAgent yes
    
    # Optional: Keep the connection alive by sending null packets every X seconds
    # ServerAliveInterval 60
    # ServerAliveCountMax 3
    
    # Optional: Disable strict host key checking for initial connection (not recommended for production)
    # StrictHostKeyChecking no
    # UserKnownHostsFile /dev/null
